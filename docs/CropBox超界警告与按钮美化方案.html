为什么追求“极致空间占用”对本窗体至关重要？ AutoScroll=false 严格受限架构

在当前项目架构中，pnlStationContent 采用 FlowLayoutPanel(WrapContents=false, AutoScroll=false)，固定总高度仅 466px。原版的 pnlPageBoxChoiceCard 强行塞入 84px，直接导致下方的预设卡片和主操作按钮被推到视窗底边缘外！

因此，“0px 额外占用”或“极致单行微带（18~24px）”是最具工程价值的方案——它让异常状态下的基准切换像“原生自带”一样安静优雅，不引发任何界面跳动。

落地代码指引：方案 ZERO-1（0px 增量）如何写？

在现有的 BuildImpositionDashboardCard() 中，头部容器已包含 pnlModeButtons。只需在其右侧添加一个微型的 pnlBoxChoiceMini： 

// 在 BuildImpositionDashboardCard() 中构建顶栏右侧的微型页框切换器 (0px 额外高度增量)
private void SetupPageBoxMiniSwitchInDashboard(Panel pnlHead)
{
    pnlBoxChoiceMini = new Panel
    {
        Name = "pnlBoxChoiceMini",
        Size = new Size(132, 20),
        Location = new Point(205, 3), // 位于 [连拼 折手] 按钮组右侧的空白区域
        BackColor = Color.FromArgb(255, 251, 230), // 浅琥珀警告色
        Visible = false // 默认隐藏，仅当超界时才显示
    };

    // 微型分段按钮组: 媒体框 / 裁剪框
    btnMiniMedia = CreateMiniBoxButton("媒 80", isMedia: true);
    btnMiniCrop  = CreateMiniBoxButton("裁 80.1", isMedia: false);

    pnlBoxChoiceMini.Controls.Add(btnMiniMedia);
    pnlBoxChoiceMini.Controls.Add(btnMiniCrop);
    pnlHead.Controls.Add(pnlBoxChoiceMini);
}

// 当检测到超界时，仅需更新文本并设为 Visible = true，完全不改动 pnlStationContent！
internal void UpdatePageBoxChoiceFromInspection(PdfInspectorInfo inspection)
{
    if (page?.MediaBox == null || page.CropBox == null)
    {
        pnlBoxChoiceMini.Visible = false;
        return;
    }

    btnMiniMedia.Text = $"媒 {Math.Round(page.MediaBox.Width * 25.4 / 72)}";
    btnMiniCrop.Text  = $"裁 {Math.Round(page.CropBox.Width * 25.4 / 72, 1)}";
    pnlBoxChoiceMini.Visible = true;
}
